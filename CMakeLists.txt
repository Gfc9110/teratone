cmake_minimum_required(VERSION 3.20)
project(OpenDaw VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${CONFIG} "${CMAKE_BINARY_DIR}/lib")
endforeach()
option(OD_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(OD_THREAD_SANITIZE "Enable ThreadSanitizer (separate build)" OFF)
if(OD_SANITIZE AND OD_THREAD_SANITIZE)
  message(FATAL_ERROR "Use separate ASan/UBSan and TSan builds")
endif()
if(NOT MSVC)
  if(OD_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  elseif(OD_THREAD_SANITIZE)
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
  endif()
endif()
option(OD_MEASURE_RENDER "Opt-in bounded block timing; measurement builds only" OFF)
if(OD_MEASURE_RENDER)
  add_compile_definitions(OD_MEASURE_RENDER=1)
endif()
find_package(Threads REQUIRED)
function(od_warnings TARGET)
  if(MSVC)
    target_compile_options(${TARGET} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
add_library(open_daw_core STATIC src/engine.cpp src/wav.cpp)
target_include_directories(open_daw_core PUBLIC include)
target_link_libraries(open_daw_core PUBLIC Threads::Threads)
od_warnings(open_daw_core)
add_library(open_daw_engine STATIC src/service.cpp src/media.cpp src/drums.cpp src/device.cpp)
target_link_libraries(open_daw_engine PUBLIC open_daw_core Threads::Threads)
od_warnings(open_daw_engine)
if(APPLE)
  target_link_libraries(open_daw_engine PUBLIC "-framework AudioUnit" "-framework AudioToolbox" "-framework CoreAudio")
elseif(WIN32)
  target_link_libraries(open_daw_engine PUBLIC ole32 uuid avrt)
endif()
add_library(open_daw_api_static STATIC src/api.cpp)
target_link_libraries(open_daw_api_static PUBLIC open_daw_engine)
od_warnings(open_daw_api_static)
add_library(open_daw_api SHARED src/api.cpp)
target_link_libraries(open_daw_api PUBLIC open_daw_engine)
target_compile_definitions(open_daw_api PRIVATE OD_SHARED OD_BUILDING)
set_target_properties(open_daw_api PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
od_warnings(open_daw_api)
add_executable(daw_demo src/main.cpp)
target_link_libraries(daw_demo PRIVATE open_daw_core)
if(WIN32)
  target_sources(daw_demo PRIVATE src/wasapi.cpp)
  target_link_libraries(daw_demo PRIVATE ole32 uuid)
endif()
add_executable(daw_session src/session_main.cpp)
target_link_libraries(daw_session PRIVATE open_daw_engine)
od_warnings(daw_session)
include(CTest)
if(BUILD_TESTING)
  add_executable(open_daw_tests tests/tests.cpp)
  target_link_libraries(open_daw_tests PRIVATE open_daw_core)
  add_test(NAME original_audio_core COMMAND open_daw_tests)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/service_tests.cpp")
    add_executable(service_tests tests/service_tests.cpp)
    target_link_libraries(service_tests PRIVATE open_daw_api_static)
    od_warnings(service_tests)
    add_test(NAME interactive_service COMMAND service_tests)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/api_smoke.c")
    add_executable(api_smoke tests/api_smoke.c)
    target_link_libraries(api_smoke PRIVATE open_daw_api_static)
    add_test(NAME pure_c_abi COMMAND api_smoke)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(capacity_tests tests/capacity_tests.cpp)
  target_link_libraries(capacity_tests PRIVATE open_daw_api_static)
  od_warnings(capacity_tests)
  add_test(NAME sixty_four_track_capacity COMMAND capacity_tests)
  add_executable(drum_tests tests/drum_tests.cpp)
  target_link_libraries(drum_tests PRIVATE open_daw_api_static)
  od_warnings(drum_tests)
  add_test(NAME internal_drum_kit COMMAND drum_tests)
  add_executable(synth_tests tests/synth_tests.cpp)
  target_link_libraries(synth_tests PRIVATE open_daw_api_static)
  od_warnings(synth_tests)
  add_test(NAME per_track_synth COMMAND synth_tests)
  add_executable(allocation_probe tests/allocation_probe.cpp)
  target_link_libraries(allocation_probe PRIVATE open_daw_engine)
  add_test(NAME callback_cpp_allocations COMMAND allocation_probe)
  if(APPLE AND NOT OD_SANITIZE AND NOT OD_THREAD_SANITIZE)
    add_library(rt_heap_audit SHARED tests/rt_heap_audit.cpp)
    add_test(NAME callback_runtime_audit COMMAND ${CMAKE_COMMAND} -E env
      "DYLD_INSERT_LIBRARIES=$<TARGET_FILE:rt_heap_audit>" OD_REQUIRE_RT_AUDIT=1 $<TARGET_FILE:allocation_probe>)
  endif()
endif()

if(OD_MEASURE_RENDER)
  add_executable(performance_probe tests/performance_probe.cpp)
  target_link_libraries(performance_probe PRIVATE open_daw_engine)
  od_warnings(performance_probe)
endif()
